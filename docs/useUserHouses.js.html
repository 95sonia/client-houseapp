import { useState } from 'react';
import toast from 'react-hot-toast';
import { useFetch } from './useFetch';
const apiUrl = import.meta.env.VITE_API_URL_BASE;

/**
 * Custom Hook para gestionar las acciones de usuario relacionadas con las viviendas.
 * * Proporciona funcionalidades para manejar la lista de favoritos (añadir, eliminar, toggle)
 * y para realizar reservas de propiedades.
 * * @returns {Object} Herramientas de interacción para el usuario:
 * @returns {Function} return.addFavorito - Añade una vivienda a la lista de favoritos.
 * @returns {Function} return.deleteFavorito - Elimina una vivienda de favoritos.
 * @returns {Function} return.toggleFavorito - Alterna entre añadir o eliminar de favoritos según el estado actual.
 * @returns {Function} return.reservarHouse - Realiza una petición de reserva para una vivienda.
 * @returns {boolean} return.loading - Indica si hay una petición fetch en curso.
 * @returns {any} return.error - Contiene información del error si la petición falla.
 */
export const useUserHouses = () => {

    const { loading, error, consultaFetch } = useFetch(); //extraer las herramientas del useFetch

    /**
     * Añade una vivienda a la lista de favoritos del usuario autenticado.
     * @async
     * @param {string} houseId - ID único de la vivienda a añadir.
     * @returns {Promise<boolean>} True si se añadió correctamente, false si hubo un error.
     */
    const addFavorito = async (houseId) => {
        try {
            // Usamos tu consultaFetch universal
            const data = await consultaFetch(`${apiUrl}/user/favoritos/${houseId}`, 'POST');

            // Si llega aquI, es que respuesta.ok true
            toast.success(data.msg || "Vivienda añadida a favoritos");
            return true;
        } catch (err) {
            // El error ya fue capturado y relanzado por useFetch
            console.log(err)
            toast.error(err.message); // Opcional
            return false;
        }
    };

    /**
     * Elimina una vivienda de la lista de favoritos del usuario autenticado.
     * @async
     * @param {string} houseId - ID único de la vivienda a eliminar.
     * @returns {Promise<boolean>} True si se eliminó correctamente.
     */
    const deleteFavorito = async (houseId) => {
        try {
            const data = await consultaFetch(`${apiUrl}/user/favoritos/${houseId}`, 'DELETE');
            toast.success(data.msg || "Eliminado de favoritos");
            return true;
        } catch (err) {
            console.log(err)
            return false;
        }
    };

    /**
     * Función lógica que decide si añadir o quitar una vivienda de favoritos.
     * @async
     * @param {string} houseId - ID único de la vivienda.
     * @param {boolean} esFavoritoYa - Estado actual de la vivienda en la interfaz (si ya es favorita o no).
     * @returns {Promise<boolean>} Resultado de la operación ejecutada.
     */
    const toggleFavorito = async (houseId, esFavoritoYa) => {
        if (esFavoritoYa) {
            return await deleteFavorito(houseId);
        } else {
            return await addFavorito(houseId);
        }
    };

    /**
     * Registra una reserva para una vivienda específica.
     * @async
     * @param {string} houseId - ID único de la vivienda a reservar.
     * @param {Object} reservaData - Datos de la reserva (fechas, número de huéspedes, etc.).
     * @returns {Promise<boolean>} True si la reserva se completó con éxito.
     */
    const reservarHouse = async (houseId, reservaData) => {
        try {
            // pasar reservaData como el body
            const data = await consultaFetch(
                `${apiUrl}/user/reservar/${houseId}`,
                'POST',
                reservaData
            );

            toast.success(data.msg || "Reserva realizada con éxito");
            return true;
        } catch (err) {
            toast.error(err.message || "Error al realizar la reserva");
            return false;
        }
    };

    return {
        addFavorito,
        deleteFavorito,
        toggleFavorito,
        reservarHouse,
        loading,
        error
    };
};
